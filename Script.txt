local function SafeDisconnect(conn)
    if conn and conn.Connected then
        conn:Disconnect()
    end
end

local function SafeRemoveDrawing(drawing)
    if not drawing then return end
    -- the method lookup itself lives inside the pcall (some executors error on a missing member)
    local ok = pcall(function() drawing:Remove() end)
    if not ok then
        pcall(function() drawing:Destroy() end)
    end
end

-- Shortens a string so it always fits `maxChars` monospace characters (".." marks a cut)
local function FitText(text, maxChars)
    text = tostring(text)
    if maxChars and maxChars > 4 and #text > maxChars then
        return string.sub(text, 1, maxChars - 2) .. ".."
    end
    return text
end

-- Shadowed property write: touches the Drawing only when the value really changed.
-- `slot` names the property inside def.Cache (any unique short string per property).
local function Put(def, slot, obj, prop, value)
    local cache = def.Cache
    if not cache then
        cache = {}
        def.Cache = cache
    end
    if cache[slot] ~= value then
        cache[slot] = value
        obj[prop] = value
    end
end

-- Health fraction (0..1) from raw Humanoid values. Safe for NaN, 0 / negative / infinite MaxHealth.
local function ComputeHpRatio(hp, maxHp)
    if type(hp) ~= "number" or hp ~= hp or hp <= 0 then return 0 end
    if type(maxHp) ~= "number" or maxHp ~= maxHp or maxHp <= 0 or maxHp == math.huge or hp == math.huge then
        return 1
    end
    return math.clamp(hp / maxHp, 0, 1)
end

local function SnapStep(val, minVal, maxVal, step)
    local clamped = math.clamp(val, minVal, maxVal)
    if not step or step <= 0 then
        return clamped
    end
    local snapped = minVal + math.floor(((clamped - minVal) / step) + 0.5) * step
    snapped = math.floor(snapped * 1e6 + 0.5) / 1e6 -- kill float drift (0.55000001)
    return math.clamp(snapped, minVal, maxVal)
end

local function To255(x)
    return math.clamp(math.floor(x * 255 + 0.5), 0, 255)
end

-- Wraps an event handler: an error inside can never break the connection chain,
-- and warnings are rate-limited (max one per tag every 5 seconds).
local GuardWarnAt = {}
local function Guard(tag, fn)
    return function(...)
        local ok, err = pcall(fn, ...)
        if not ok then
            local now = os.clock()
            if (GuardWarnAt[tag] or -10) + 5 < now then
                GuardWarnAt[tag] = now
                warn(string.format("[0x] %s error: %s", tag, tostring(err)))
            end
        end
    end
end

local function BindLocalCharacter(char)
    for _, conn in pairs(WorldState.LocalCharConns) do
        SafeDisconnect(conn)
    end
    table.clear(WorldState.LocalCharConns)

    WorldState.LocalChar = char
    WorldState.LocalRoot = nil
    if not char then return end

    local root = char:FindFirstChild("HumanoidRootPart")
    if root and root:IsA("BasePart") then
        WorldState.LocalRoot = root
    end

    WorldState.LocalCharConns.ChildAdded = char.ChildAdded:Connect(function(child)
        if child.Name == "HumanoidRootPart" and child:IsA("BasePart") then
            WorldState.LocalRoot = child
        end
    end)

    WorldState.LocalCharConns.ChildRemoved = char.ChildRemoved:Connect(function(child)
        if child == WorldState.LocalRoot then
            WorldState.LocalRoot = nil
        end
    end)
end

local function BindCamera(cam)
    SafeDisconnect(WorldState.CameraConn)
    WorldState.CameraConn = nil
    WorldState.Camera = cam
    if not cam then return end

    WorldState.ViewportSize = SanitizeViewport(cam.ViewportSize)
    WorldState.CameraConn = cam:GetPropertyChangedSignal("ViewportSize"):Connect(function()
        WorldState.ViewportSize = SanitizeViewport(cam.ViewportSize)
        if Crosshair then
            Crosshair:MarkDirty()
        end
        if UI then
            UI:ClampToViewport()
        end
    end)

    if Crosshair then
        Crosshair:MarkDirty()
    end
end

function ConfigIO.IsSupported()
    return writefile ~= nil and readfile ~= nil and isfile ~= nil
end

function ConfigIO.FileFor(slot)
    if slot == 1 then
        return ConfigIO.BaseName .. ".json"
    end
    return string.format("%s_%d.json", ConfigIO.BaseName, slot)
end

function ConfigIO.ValidateColorTable(tbl)
    if type(tbl) ~= "table" then return nil end
    local r, g, b = tonumber(tbl.r), tonumber(tbl.g), tonumber(tbl.b)
    if not r or not g or not b then return nil end
    if r ~= r or g ~= g or b ~= b then return nil end
    if math.abs(r) == math.huge or math.abs(g) == math.huge or math.abs(b) == math.huge then return nil end
    return Color3.fromRGB(
        math.clamp(math.floor(r + 0.5), 0, 255),
        math.clamp(math.floor(g + 0.5), 0, 255),
        math.clamp(math.floor(b + 0.5), 0, 255)
    )
end

local function IsBindableKey(kc)
    return typeof(kc) == "EnumItem" and kc.EnumType == Enum.KeyCode and not RESERVED_KEYS[kc.Name]
end

-- Returns a validated value or nil
function ConfigIO.ParseValue(rule, raw)
    local t = rule.type
    if t == "boolean" then
        if type(raw) == "boolean" then return raw end
    elseif t == "number" then
        local n = tonumber(raw)
        if n and n == n and math.abs(n) ~= math.huge then
            return SnapStep(n, rule.min, rule.max, rule.step)
        end
    elseif t == "color" then
        if typeof(raw) == "Color3" then return raw end
        return ConfigIO.ValidateColorTable(raw)
    elseif t == "keycode" then
        if typeof(raw) == "EnumItem" then
            if IsBindableKey(raw) then
                return raw
            end
            return nil
        end
        if type(raw) == "string" and #raw <= 32 then
            local ok, kc = pcall(function() return Enum.KeyCode[raw] end)
            if ok and IsBindableKey(kc) then
                return kc
            end
        end
    end
    return nil
end

-- Integrity audit of the LIVE config against the schema. Returns (badCount, totalCount).
function ConfigIO.Audit()
    local bad, total = 0, 0
    if Config.UnloadKey == Config.MenuToggleKey then
        bad += 1 -- both hotkeys on the same key: the kill switch would be ambiguous
    end
    for _, rule in ipairs(ConfigSchema) do
        total += 1
        local cur = Config[rule.key]
        if rule.type == "number" then
            if type(cur) ~= "number" then
                bad += 1
            else
                local parsed = ConfigIO.ParseValue(rule, cur)
                if parsed == nil or math.abs(parsed - cur) > 1e-4 then
                    bad += 1
                end
            end
        else
            if ConfigIO.ParseValue(rule, cur) == nil then
                bad += 1
            end
        end
    end
    return bad, total
end

function ConfigIO.Serialize(rule, value)
    if rule.type == "color" then
        return ColorToRGBTable(value)
    elseif rule.type == "keycode" then
        return value.Name
    end
    return value
end

end

-- Canonical text of a full config snapshot (stable key order) -> integrity checksum
function ConfigIO.Canonical(values)
    local parts = table.create(#ConfigSchema)
    for i, rule in ipairs(ConfigSchema) do
        local v = values[rule.key]
        local s
        if rule.type == "color" and typeof(v) == "Color3" then
            s = string.format("%d,%d,%d", To255(v.R), To255(v.G), To255(v.B))
        elseif rule.type == "keycode" and typeof(v) == "EnumItem" then
            s = v.Name
        elseif rule.type == "number" and type(v) == "number" then
            s = string.format("%.3f", v)
        else
            s = tostring(v)
        end
        parts[i] = rule.key .. "=" .. s
    end
    return table.concat(parts, ";")
end

-- Not cryptographic: detects corruption / hand edits, never used for trust decisions
function ConfigIO.Checksum(values)
    local text = ConfigIO.Canonical(values)
    local h = 5381
    for i = 1, #text do
        h = (h * 33 + string.byte(text, i)) % 4294967296
    end
    return string.format("%08x", math.floor(h))
end

-- Repairs the LIVE config in place (wrong type / out of range / invalid key). Returns the number of fixed keys.
function ConfigIO.Repair()
    local fixed = 0
    for _, rule in ipairs(ConfigSchema) do
        local cur = Config[rule.key]
        local parsed = ConfigIO.ParseValue(rule, cur)
        local bad
        if rule.type == "number" then
            bad = type(cur) ~= "number" or parsed == nil or math.abs(parsed - cur) > 1e-4
        else
            bad = parsed == nil
        end
        if bad then
            if rule.type == "number" and parsed ~= nil then
                Config[rule.key] = parsed -- clamped / snapped into range
            else
                Config[rule.key] = DefaultConfig[rule.key]
            end
            fixed += 1
        end
    end
    if Config.UnloadKey == Config.MenuToggleKey then
        Config.UnloadKey = DefaultConfig.UnloadKey
        Config.MenuToggleKey = DefaultConfig.MenuToggleKey
        fixed += 1
    end
    if fixed > 0 then
        AfterConfigChange()
    end
    return fixed
end

local function SafeRead(name)
    if not ConfigIO.IsSupported() then return nil, "NO FILE API" end
    local okEx, exists = pcall(isfile, name)
    if not okEx or not exists then return nil, "MISSING" end
    local okRead, raw = pcall(readfile, name)
    if not okRead or type(raw) ~= "string" then return nil, "READ ERR" end
    return raw
end

function ConfigIO.Decode(raw)
    if type(raw) ~= "string" or #raw == 0 or #raw > ConfigIO.MaxFileSize then
        return nil, "CORRUPT FILE"
    end
    local ok, decoded = pcall(HttpService.JSONDecode, HttpService, raw)
    if not ok or type(decoded) ~= "table" then
        return nil, "CORRUPT FILE"
    end
    if decoded.version ~= nil then
        if type(decoded.version) ~= "number" or decoded.version > ConfigIO.FormatVersion then
            return nil, "NEWER FORMAT"
        end
    end
    return decoded
end

-- Builds a full validated config snapshot. Missing/invalid keys fall back to defaults.
function ConfigIO.Stage(decoded)
    local settingsTbl = type(decoded.settings) == "table" and decoded.settings or decoded
    local staged, applied, skipped = {}, 0, 0

    for _, rule in ipairs(ConfigSchema) do
        local key = rule.key
        local rawVal = settingsTbl[key]
        local value = nil
        if rawVal ~= nil then
            value = ConfigIO.ParseValue(rule, rawVal)
            if value ~= nil then
                applied += 1
            else
                skipped += 1
            end
        end
        if value == nil then
            value = DefaultConfig[key]
        end
        staged[key] = value
    end

    -- Migration (v10.x -> v11.x): the dot did not have its own color, inherit the crosshair look
    if settingsTbl.ColorDot == nil then
        staged.ColorDot = staged.ColorCrosshair
    end
    if settingsTbl.ColorDotAlpha == nil then
        staged.ColorDotAlpha = staged.ColorCrosshairAlpha
    end

    if staged.UnloadKey == staged.MenuToggleKey then
        staged.UnloadKey = DefaultConfig.UnloadKey
        staged.MenuToggleKey = DefaultConfig.MenuToggleKey
    end

    return staged, applied, skipped
end

function ConfigIO.SaveSlotMeta()
    if not ConfigIO.IsSupported() then return end
    pcall(writefile, ConfigIO.SlotMetaFile, tostring(ConfigIO.Slot))
end

function ConfigIO.LoadSlotMeta()
    local raw = SafeRead(ConfigIO.SlotMetaFile)
    if raw then
        local n = tonumber(string.match(raw, "%d+"))
        if n then
            n = math.floor(n)
            if n >= 1 and n <= ConfigIO.MaxSlots then
                ConfigIO.Slot = n
            end
        end
    end
end

-- Full validated snapshot of the live config. Never throws: a bad value falls back to its default.
function ConfigIO.BuildPayload(includeMenu)
    local settings = {}
    for _, rule in ipairs(ConfigSchema) do
        local okS, val = pcall(ConfigIO.Serialize, rule, Config[rule.key])
        if not okS or val == nil then
            val = ConfigIO.Serialize(rule, DefaultConfig[rule.key])
        end
        settings[rule.key] = val
    end

    return {
        version = ConfigIO.FormatVersion,
        script = VERSION,
        build = BUILD,
        checksum = ConfigIO.Checksum(Config),
        menu = (includeMenu and UI and UI.Ready) and {
            x = math.floor(UI.Position.X),
            y = math.floor(UI.Position.Y),
        } or nil,
        settings = settings,
    }
end

local function SaveFailed(reason)
    Health.LastSaveOk = false
    Health.SaveFailures += 1
    return false, reason
end

function ConfigIO.Save()
    if not ConfigIO.IsSupported() then
        return false, "NO FILE API"
    end

    local payload = ConfigIO.BuildPayload(true)
    local okEncode, encoded = pcall(HttpService.JSONEncode, HttpService, payload)
    if not okEncode or type(encoded) ~= "string" or #encoded > ConfigIO.MaxFileSize then
        return SaveFailed("ENCODE ERR")
    end

    local fileName = ConfigIO.FileFor(ConfigIO.Slot)
    local oldRaw = SafeRead(fileName)

    -- Nothing changed since the last save: skip the disk write entirely
    if oldRaw == encoded then
        Health.LastSaveOk = true
        ConfigIO.Dirty = false
        return true, string.format("SAVED #%d", ConfigIO.Slot)
    end

    -- Keep the last known-good file as a backup before overwriting
    if oldRaw and ConfigIO.Decode(oldRaw) then
        pcall(writefile, fileName .. ".bak", oldRaw)
    end

    local okWrite = pcall(writefile, fileName, encoded)
    if not okWrite then
        return SaveFailed("WRITE ERR")
    end

    -- Read-back verification: a silent executor write failure must never go unnoticed
    -- (an executor may normalise line endings, so a differing text is accepted when it carries OUR checksum)
    local back = SafeRead(fileName)
    local backOk = back == encoded
    if not backOk and back then
        local backDecoded = ConfigIO.Decode(back)
        backOk = backDecoded ~= nil and backDecoded.checksum == payload.checksum
    end
    if not backOk then
        if oldRaw then
            pcall(writefile, fileName, oldRaw) -- put the previous good file back
        end
        return SaveFailed("VERIFY ERR")
    end

    Health.LastSaveOk = true
    Health.SaveCount += 1
    ConfigIO.Dirty = false
    return true, string.format("SAVED #%d", ConfigIO.Slot)
end

-- Validates a decoded table and applies it atomically (all keys or nothing).
-- Returns ok, skippedCount, editedFlag
function ConfigIO.Apply(decoded, isInitialStartup)
    local staged, appliedCount, skippedCount = ConfigIO.Stage(decoded)
    if appliedCount == 0 then
        return false, 0, false
    end

    local edited = type(decoded.checksum) == "string" and decoded.checksum ~= ConfigIO.Checksum(staged)

    for k, v in pairs(staged) do
        Config[k] = v
    end

    if isInitialStartup and UI and UI.Ready and type(decoded.menu) == "table" then
        local mx = tonumber(decoded.menu.x)
        local my = tonumber(decoded.menu.y)
        if mx and my and mx == mx and my == my and math.abs(mx) ~= math.huge and math.abs(my) ~= math.huge then
            local viewport = WorldState.ViewportSize
            local clampedX = math.clamp(math.floor(mx), 0, math.max(0, viewport.X - UI.Width))
            local clampedY = math.clamp(math.floor(my), 0, math.max(0, viewport.Y - UI.Height))
            UI.Position = Vector2.new(clampedX, clampedY)
            UI:UpdateLayout()
        end
    end

    AfterConfigChange()
    return true, skippedCount, edited
end

function ConfigIO.Load(isInitialStartup)
    if not ConfigIO.IsSupported() then
        return false, "NO FILE API"
    end

    local fileName = ConfigIO.FileFor(ConfigIO.Slot)
    local raw, readErr = SafeRead(fileName)
    local decoded, decodeErr = nil, readErr
    local usedBackup = false

    if raw then
        decoded, decodeErr = ConfigIO.Decode(raw)
    end

    if not decoded then
        local bakRaw = SafeRead(fileName .. ".bak")
        if bakRaw then
            decoded = ConfigIO.Decode(bakRaw)
            usedBackup = decoded ~= nil
        end
    end

    if not decoded then
        if decodeErr == "MISSING" then
            return false, "NO SAVE FILE"
        end
        Health.LastLoadOk = false
        return false, decodeErr or "CORRUPT FILE"
    end

    local ok, skippedCount, edited = ConfigIO.Apply(decoded, isInitialStartup)
    if not ok then
        Health.LastLoadOk = false
        return false, "CORRUPT FILE"
    end
    Health.LastLoadOk = true

    if usedBackup then
        return true, string.format("LOADED #%d (BAK)", ConfigIO.Slot)
    end
    if skippedCount > 0 then
        return true, string.format("LOADED #%d (%d SKIP)", ConfigIO.Slot, skippedCount)
    end
    if edited then
        return true, string.format("LOADED #%d (EDITED)", ConfigIO.Slot)
    end
    return true, string.format("LOADED #%d", ConfigIO.Slot)
end

-- Copies the live config (without menu position) to the clipboard as JSON
function ConfigIO.Export()
    if not setclipboard then
        return false, "NO CLIPBOARD API"
    end
    local okEncode, encoded = pcall(HttpService.JSONEncode, HttpService, ConfigIO.BuildPayload(false))
    if not okEncode or type(encoded) ~= "string" then
        return false, "ENCODE ERR"
    end
    local okCopy = pcall(setclipboard, encoded)
    if not okCopy then
        return false, "COPY ERR"
    end
    return true, "COPIED TO CLIPBOARD"
end

-- Imports a config from raw JSON text (validated exactly like a file load)
function ConfigIO.Import(raw)
    if type(raw) == "string" then
        raw = string.match(raw, "^%s*(.-)%s*$")
    end
    local decoded = ConfigIO.Decode(raw)
    if not decoded then
        return false, "BAD CONFIG TEXT"
    end
    local ok, skipped = ConfigIO.Apply(decoded, false)
    if not ok then
        return false, "NOTHING VALID"
    end
    if skipped > 0 then
        return true, string.format("IMPORTED (%d SKIP)", skipped)
    end
    return true, "IMPORTED"
end

-- Clipboard first (when the executor can read it), then the 0x_import.json file
function ConfigIO.ImportAuto()
    if getclipboard then
        local okClip, text = pcall(getclipboard)
        if okClip and type(text) == "string" and ConfigIO.Decode((string.match(text, "^%s*(.-)%s*$"))) then
            return ConfigIO.Import(text)
        end
    end
    local fileRaw = SafeRead(ConfigIO.ImportFile)
    if fileRaw then
        return ConfigIO.Import(fileRaw)
    end
    if not getclipboard then
        return false, "PUT 0x_import.json"
    end
    return false, "NO CONFIG FOUND"
end

function ConfigIO.Reset()
    for k, v in pairs(DefaultConfig) do
        Config[k] = v
    end
    AfterConfigChange()
    return true, "DEFAULTS RESTORED"
end

function ConfigIO.SetSlot(n)
    n = math.floor(tonumber(n) or 0)
    if n < 1 or n > ConfigIO.MaxSlots then
        return false, "BAD SLOT"
    end
    if n == ConfigIO.Slot then
        return true, string.format("SLOT %d ACTIVE", n)
    end
    -- A pending auto-save belongs to the slot the user was editing: flush it BEFORE switching
    if ConfigIO.Dirty and Config.AutoSave then
        ConfigIO.Save()
    end
    ConfigIO.Dirty = false
    ConfigIO.Slot = n
    ConfigIO.SaveSlotMeta()
    local ok, msg = ConfigIO.Load(false)
    if not ok and msg == "NO SAVE FILE" then
        return true, string.format("SLOT %d EMPTY", n)
    end
    return ok, msg
end

-- Validated setter used by the public API
function ConfigIO.Set(key, value)
    local rule = ConfigIO.Rules[key]
    if not rule then return false, "UNKNOWN KEY" end
    local parsed = ConfigIO.ParseValue(rule, value)
    if parsed == nil then return false, "INVALID VALUE" end
    if rule.type == "keycode" then
        local other = (key == "UnloadKey") and Config.MenuToggleKey or Config.UnloadKey
        if parsed == other then return false, "KEY IN USE" end
    end
    Config[key] = parsed
    AfterConfigChange()
    ConfigIO.NotifyChanged()
    return true, "OK"
end

-- Marks the config dirty. The actual write is done by ConfigIO.Tick (frame driven, no threads).
function ConfigIO.NotifyChanged()
    if IsUnloaded or not ConfigIO.IsSupported() then return end
    ConfigIO.Dirty = true
    ConfigIO.DirtyAt = os.clock()
    ConfigIO.Retries = 0
end

-- Debounced auto-save: fires 1.5s after the last change, retries up to 3 times with a 5s back-off
function ConfigIO.Tick(now)
    if not ConfigIO.Dirty then return end
    if not Config.AutoSave then
        ConfigIO.Dirty = false
        return
    end
    if (now - ConfigIO.DirtyAt) < 1.5 then return end

    ConfigIO.Dirty = false
    local ok, msg = ConfigIO.Save()
    if ok then
        if UI and UI.Ready then UI:SetStatus("AUTO-SAVED") end
    else
        ConfigIO.Retries += 1
        if ConfigIO.Retries <= 3 then
            ConfigIO.Dirty = true
            ConfigIO.DirtyAt = now + 5
        end
        if UI and UI.Ready then UI:SetStatus("AUTOSAVE: " .. tostring(msg)) end
    end
end

    if ActiveBanner then

        end
    end

-- 10. Dirty-Flagged Pixel Crosshair (Lines + Fully Independent Dot)
--==============================================================================
Crosshair = {
    Objects = {},
    LinesOutline = {},
    LinesInner = {},
    DotOutline = nil,
    DotInner = nil,
    LinesShown = false,
    DotShown = false,
    Dirty = true,
    Ready = false,
}

function Crosshair:NewDrawing(className, props)
    local obj = Drawing.new(className)
    table.insert(self.Objects, obj)
    for k, v in pairs(props) do
        obj[k] = v
    end
    return obj
end

function Crosshair:MarkDirty()
    self.Dirty = true
    self:Flush()
end

function Crosshair:Init()
    local ok, err = pcall(function()
        for i = 1, 4 do
            self.LinesOutline[i] = self:NewDrawing("Line", {
                Thickness = 3,
                Color = Color3.fromRGB(0, 0, 0),
                Transparency = 1,
                ZIndex = 8,
                Visible = false,
            })
            self.LinesInner[i] = self:NewDrawing("Line", {
                Thickness = 1,
                Color = Config.ColorCrosshair,
                Transparency = 1,
                ZIndex = 9,
                Visible = false,
            })
        end

        self.DotOutline = self:NewDrawing("Square", {
            Size = Vector2.new(4, 4),
            Color = Color3.fromRGB(0, 0, 0),
            Filled = true,
            Transparency = 1,
            ZIndex = 8,
            Visible = false,
        })

        self.DotInner = self:NewDrawing("Square", {
            Size = Vector2.new(2, 2),
            Color = Config.ColorDot,
            Filled = true,
            Transparency = 1,
            ZIndex = 9,
            Visible = false,
        })
    end)

    if ok then
        self.Ready = true
        self:MarkDirty()
    else
        warn("[0x] Crosshair failed to initialize: " .. tostring(err))
    end
end

function Crosshair:HideLines()
    if not self.LinesShown then return end
    self.LinesShown = false
    for i = 1, 4 do
        self.LinesOutline[i].Visible = false
        self.LinesInner[i].Visible = false
    end
end

function Crosshair:HideDot()
    if not self.DotShown then return end
    self.DotShown = false
    self.DotOutline.Visible = false
    self.DotInner.Visible = false
end

function Crosshair:Hide()
    if not self.Ready then return end
    self:HideLines()
    self:HideDot()
end

function Crosshair:Flush()
    if not self.Dirty or IsUnloaded or not self.Ready then
        return
    end
    self.Dirty = false

    local viewport = WorldState.ViewportSize
    local cx = math.floor(viewport.X * 0.5)
    local cy = math.floor(viewport.Y * 0.5)

    --------------------------------------------------------------------------
    -- Lines (controlled ONLY by ShowCrosshair / size / gap / ColorCrosshair)
    --------------------------------------------------------------------------
    local size = math.floor(Config.CrosshairSize)
    local gap = math.floor(Config.CrosshairGap)

    if Config.ShowCrosshair and size > 0 then
        local col = Config.ColorCrosshair
        local alpha = Config.ColorCrosshairAlpha
        local lo, li = self.LinesOutline, self.LinesInner

        -- Top
        lo[1].From = Vector2.new(cx, cy - gap + 1)
        lo[1].To   = Vector2.new(cx, cy - gap - size - 1)
        li[1].From = Vector2.new(cx, cy - gap)
        li[1].To   = Vector2.new(cx, cy - gap - size)

        -- Bottom
        lo[2].From = Vector2.new(cx, cy + gap - 1)
        lo[2].To   = Vector2.new(cx, cy + gap + size + 1)
        li[2].From = Vector2.new(cx, cy + gap)
        li[2].To   = Vector2.new(cx, cy + gap + size)

        -- Left
        lo[3].From = Vector2.new(cx - gap + 1, cy)
        lo[3].To   = Vector2.new(cx - gap - size - 1, cy)
        li[3].From = Vector2.new(cx - gap, cy)
        li[3].To   = Vector2.new(cx - gap - size, cy)

        -- Right
        lo[4].From = Vector2.new(cx + gap - 1, cy)
        lo[4].To   = Vector2.new(cx + gap + size + 1, cy)
        li[4].From = Vector2.new(cx + gap, cy)
        li[4].To   = Vector2.new(cx + gap + size, cy)

        local reveal = not self.LinesShown
        for i = 1, 4 do
            li[i].Color = col
            li[i].Transparency = alpha
            lo[i].Transparency = alpha
            if reveal then
                lo[i].Visible = true
                li[i].Visible = true
            end
        end
        self.LinesShown = true
    else
        self:HideLines()
    end

    --------------------------------------------------------------------------
    -- Dot (controlled ONLY by CrosshairDot / dot size / ColorDot / ColorDotAlpha)
    --------------------------------------------------------------------------
    if Config.CrosshairDot then
        local d = math.clamp(math.floor(Config.CrosshairDotSize), 1, 8)
        local half = math.floor(d / 2)
        local ix, iy = cx - half, cy - half
        local dotAlpha = Config.ColorDotAlpha

        self.DotOutline.Size = Vector2.new(d + 2, d + 2)
        self.DotOutline.Position = Vector2.new(ix - 1, iy - 1)
        self.DotInner.Size = Vector2.new(d, d)
        self.DotInner.Position = Vector2.new(ix, iy)
        self.DotInner.Color = Config.ColorDot
        self.DotInner.Transparency = dotAlpha
        self.DotOutline.Transparency = dotAlpha
        if not self.DotShown then
            self.DotOutline.Visible = true
            self.DotInner.Visible = true
            self.DotShown = true
        end
    else
        self:HideDot()
    end
end

function Crosshair:Destroy()
    self.Ready = false
    for _, obj in ipairs(self.Objects) do
        SafeRemoveDrawing(obj)
    end
    table.clear(self.Objects)
    table.clear(self.LinesOutline)
    table.clear(self.LinesInner)
    self.DotOutline = nil
    self.DotInner = nil
    self.LinesShown = false
    self.DotShown = false
end

local function CleanName(player)
    local n = player.DisplayName
    if type(n) ~= "string" or n == "" then
        n = player.Name
    end
    n = (string.gsub(n, "[^ -~]", "")) -- printable ASCII only (Drawing fonts are not unicode-safe)
    n = string.sub(n, 1, 14)
    if #n == 0 then n = "?" end
    return n
end

function Combat:Update(now, count, visCount, nearest, nearestDistSq, nearestOn, nx, ny)
    if not self.Ready or IsUnloaded then return end

    local vp = WorldState.ViewportSize
    if vp ~= self.LastViewport then
        self.LastViewport = vp
        self:Layout()
    end

    -- Threat HUD (text refreshed at 10 Hz, only touches Drawings when the text changed)
    if Config.ThreatHUD then
        self:SetHudVisible(true)
        if (now - self.LastUpdate) >= 0.1 then
            self.LastUpdate = now
            if Config.VisibilityCheck then
                self:SetText(2, string.format("ENEMIES : %d (%d VIS)", count, visCount))
            else
                self:SetText(2, string.format("ENEMIES : %d", count))
            end
            if nearest and nearest.Player and not nearest.Destroyed then
                -- health comes from the event-driven cache (no per-frame Humanoid reads)
                local hp = "--"
                if nearest.HpKnown then
                    hp = string.format("%d%%", math.clamp(math.floor(nearest.HpRatio * 100 + 0.5), 0, 100))
                end
                self:SetText(3, "NEAREST : " .. CleanName(nearest.Player))
                self:SetText(4, string.format("DIST/HP : %dm / %s", math.floor(math.sqrt(nearestDistSq) + 0.5), hp))
            else
                self:SetText(3, "NEAREST : --")
                self:SetText(4, "DIST/HP : --")
            end
        end
    else
        self:SetHudVisible(false)
    end

    -- Threat line (screen center -> nearest on-screen enemy)
    if Config.ThreatLine and nearest and nearestOn then
        local color = nearest.LastColor or Config.ColorHidden
        local alpha = nearest.LastAlpha or Config.ColorHiddenAlpha
        local line = self.ThreatLine
        local fx, fy = math.floor(vp.X * 0.5), math.floor(vp.Y * 0.5)
        local tx, ty = math.floor(nx), math.floor(ny)
        if self.LineFX ~= fx or self.LineFY ~= fy or self.LineTX ~= tx or self.LineTY ~= ty then
            self.LineFX, self.LineFY, self.LineTX, self.LineTY = fx, fy, tx, ty
            line.From = Vector2.new(fx, fy)
            line.To = Vector2.new(tx, ty)
        end
        if self.LineColor ~= color or self.LineAlpha ~= alpha then
            self.LineColor, self.LineAlpha = color, alpha
            line.Color = color
            line.Transparency = alpha
        end
        if not self.LineShown then
            line.Visible = true
            self.LineShown = true
        end
    else
        self:HideLine()
    end
end

function Combat:Destroy()
    self.Ready = false
    for _, obj in ipairs(self.Objects) do
        SafeRemoveDrawing(obj)
    end
    table.clear(self.Objects)
    table.clear(self.Lines)
    table.clear(self.Cache)
    self.HudShown = false
    self.LineShown = false
    self.LineColor, self.LineAlpha = nil, nil
end

-- 11c. Security Monitor (self-audit of THIS script: integrity, limits, runtime health)
--      Group "SEC" -> Security %, Group "REL" -> Reliability %. Add checks with Security:Register().
--==============================================================================
local COLOR_GOOD = Color3.fromRGB(0, 255, 0)
local COLOR_WARN = Color3.fromRGB(255, 220, 80)
local COLOR_BAD = Color3.fromRGB(255, 80, 80)

Security = {
    Checks = {},
    SecurityPct = 0,
    ReliabilityPct = 0,
    OverallPct = 0,
    Level = 2, -- 1 = secure, 2 = caution, 3 = warning
    Verdict = "STATUS: SCANNING",
    Advice = "PLEASE WAIT",
    LastScan = -10,
    ScanCount = 0,
}

-- 0..1 weight that fades from 1 (just happened) to 0 (older than `window` seconds)
local function Recency(since, window)
    return math.clamp(1 - (os.clock() - since) / window, 0, 1)
end

function Security:Register(def)
    def.Score = 1
    table.insert(self.Checks, def)
    return def
end

function Security:ColorFor(pct)
    if pct >= 90 then return COLOR_GOOD end
    if pct >= 70 then return COLOR_WARN end
    return COLOR_BAD
end

function Security:LevelColor()
    if self.Level == 1 then return COLOR_GOOD end
    if self.Level == 2 then return COLOR_WARN end
    return COLOR_BAD
end

function Security:Scan()
    local sums = { SEC = { 0, 0 }, REL = { 0, 0 } }
    local criticalFail = false
    local weakName, weakHint, weakShortfall = nil, nil, 0.01

    for _, chk in ipairs(self.Checks) do
        local ok, score = pcall(chk.Eval)
        if not ok or type(score) ~= "number" or score ~= score then
            score = 0
        end
        score = math.clamp(score, 0, 1)
        chk.Score = score

        local weight = chk.Weight or 1
        local bucket = sums[chk.Group] or sums.SEC
        bucket[1] += score * weight
        bucket[2] += weight

        if chk.Critical and score < 0.5 then
            criticalFail = true
        end
        local shortfall = (1 - score) * weight
        if shortfall > weakShortfall then
            weakShortfall = shortfall
            weakName = chk.Name
            weakHint = chk.Hint
        end
    end

    local sec = (sums.SEC[2] > 0) and (sums.SEC[1] / sums.SEC[2] * 100) or 100
    local rel = (sums.REL[2] > 0) and (sums.REL[1] / sums.REL[2] * 100) or 100
    local overall = sec * 0.55 + rel * 0.45
    if criticalFail then
        overall = math.min(overall, 60)
    end

    self.SecurityPct = math.clamp(math.floor(sec + 0.5), 0, 100)
    self.ReliabilityPct = math.clamp(math.floor(rel + 0.5), 0, 100)
    self.OverallPct = math.clamp(math.floor(overall + 0.5), 0, 100)
    self.LastScan = os.clock()
    self.ScanCount += 1

    if self.OverallPct >= 90 then
        self.Level = 1
        self.Verdict = string.format("OVERALL %d%%: SECURE", self.OverallPct)
        self.Advice = "NO NEED TO WORRY"
    elseif self.OverallPct >= 70 then
        self.Level = 2
        self.Verdict = string.format("OVERALL %d%%: CAUTION", self.OverallPct)
        self.Advice = "CHECK: " .. (weakHint or weakName or "SETTINGS")
    else
        self.Level = 3
        self.Verdict = string.format("OVERALL %d%%: WARNING", self.OverallPct)
        self.Advice = "FIX NOW: " .. (weakHint or weakName or "SETTINGS")
    end
end

-- Called every frame from the safe frame wrapper. Re-scans once per second while the tab is on screen and
-- every 5 seconds in the background (keeps the debug overlay and the self-repair checks current).
function Security:Tick(now)
    local onScreen = UI ~= nil and UI.Ready and UI.Visible and UI.ActiveTab == UI.TabIndex.SECURITY
    if (now - self.LastScan) < (onScreen and 1 or 5) then return end
    self:Scan()
    if onScreen then
        UI:RefreshActiveTab()
    end
end

-- Header: live clock on the right (only touches the Drawing when the second changes).
-- Uses the typed os.date("*t") overload (a runtime format string triggers the
-- 'Expected this to be "!*t" | "*t"' analyzer error).
function UI:TickClock()
    if not self.Ready or not self.Visible or not self.ClockDraw then return end
    local t = os.time()
    if t == self.LastClock then return end
    self.LastClock = t

    local text = "--:--:--"
    local ok, d = pcall(function()
        return os.date("*t", t)
    end)
    if ok and type(d) == "table" then
        text = string.format("%02d:%02d:%02d", tonumber(d.hour) or 0, tonumber(d.min) or 0, tonumber(d.sec) or 0)
    end
    Put(self, "clk", self.ClockDraw, "Text", text)
end

-- tabId: the Id string of a tab ("VISUALS", "COMBAT", ...) or its numeric index
function UI:AddControl(tabId, def)
    local handler = Controls[def.Type]
    if not handler then return nil end
    local idx = (type(tabId) == "number") and tabId or self.TabIndex[tabId]
    local tab = idx and self.Tabs[idx]
    if not tab then
        warn("[0x] Unknown tab: " .. tostring(tabId))
        return nil
    end
    def.TabIdx = idx
    handler.Create(self, def)
    table.insert(tab.Controls, def)
    return def
end

-- Shorthand: checkbox bound to a Config key. after(v) runs after the value is written.
function UI:AddToggle(tabId, text, key, after)
    return self:AddControl(tabId, {
        Type = "checkbox",
        Text = text,
        Get = function() return Config[key] end,
        Set = function(v)
            Config[key] = v
            if after then after(v) end
        end,
    })
end

-- Shorthand: slider bound to a Config key
function UI:AddSlider(tabId, text, key, min, max, step, suffix, after)
    return self:AddControl(tabId, {
        Type = "slider",
        Text = text,
        Suffix = suffix or "",
        Min = min,
        Max = max,
        Step = step,
        Get = function() return Config[key] end,
        Set = function(v)
            Config[key] = v
            if after then after(v) end
        end,
    })
end

function UI:IsTabInWindow(i)
    return i >= (self.TabOffset + 1) and i <= (self.TabOffset + self.TabWindow)
end

-- Positions the visible slice of the tab strip + the two scroll arrows
function UI:LayoutTabStrip()
    local x = math.floor(self.Position.X)
    local tabY = math.floor(self.Position.Y) + self.HeaderHeight + 1
    local areaW = self.Width - self.ArrowW * 2
    local tabW = math.floor(areaW / self.TabWindow)
    local lastW = areaW - tabW * (self.TabWindow - 1)
    local firstIdx = self.TabOffset + 1

    for i, btn in ipairs(self.TabButtons) do
        if self:IsTabInWindow(i) then
            local slot = i - firstIdx
            local w = (slot == self.TabWindow - 1) and lastW or tabW
            local bx = x + self.ArrowW + slot * tabW
            btn.Width = w
            btn.X, btn.Y, btn.W, btn.H = bx, tabY, w, self.TabBarHeight
            btn.Bg.Position = Vector2.new(bx, tabY)
            btn.Bg.Size = Vector2.new(w, self.TabBarHeight)
            btn.Indicator.Position = Vector2.new(bx + 3, tabY + self.TabBarHeight - 3)
            btn.Indicator.Size = Vector2.new(math.max(w - 6, 1), 2)
            btn.Label.Position = Vector2.new(bx + math.floor(w * 0.5), tabY + 3)
        end
    end

    local rx = x + self.Width - self.ArrowW
    self.ArrowL = { X = x, Y = tabY, W = self.ArrowW, H = self.TabBarHeight }
    self.ArrowR = { X = rx, Y = tabY, W = self.ArrowW, H = self.TabBarHeight }
    self.ArrowLBg.Position = Vector2.new(x, tabY)
    self.ArrowLText.Position = Vector2.new(x + math.floor(self.ArrowW * 0.5), tabY + 3)
    self.ArrowRBg.Position = Vector2.new(rx, tabY)
    self.ArrowRText.Position = Vector2.new(rx + math.floor(self.ArrowW * 0.5), tabY + 3)
end

function UI:UpdateLayout()
    if not self.Ready then return end
    local x = math.floor(self.Position.X)
    local y = math.floor(self.Position.Y)
    local w = self.Width
    local h = self.Height

    self.OuterBorder.Position = Vector2.new(x - 2, y - 2)
    self.OuterBorder.Size = Vector2.new(w + 4, h + 4)

    self.Background.Position = Vector2.new(x, y)
    self.Background.Size = Vector2.new(w, h)

    self.TitleText.Position = Vector2.new(x + 8, y + 5)

    -- Owner badge (next to "0x")
    local bx = x + 30
    self.OwnerOuter.Position = Vector2.new(bx, y + 5)
    self.OwnerInner.Position = Vector2.new(bx + 1, y + 6)
    self.OwnerLabelDraw.Position = Vector2.new(bx + math.floor(self.OwnerW * 0.5), y + 6)

    -- Clock (right-aligned)
    self.ClockDraw.Position = Vector2.new(x + w - 8 - self.ClockW, y + 6)

    local headerBottomY = y + self.HeaderHeight
    self.HeaderLine.From = Vector2.new(x, headerBottomY)
    self.HeaderLine.To = Vector2.new(x + w, headerBottomY)

    self:LayoutTabStrip()

    local tabBottomY = headerBottomY + 1 + self.TabBarHeight
    self.TabLine.From = Vector2.new(x, tabBottomY)
    self.TabLine.To = Vector2.new(x + w, tabBottomY)

    local contentStartX = x + self.Padding
    local contentStartY = tabBottomY + 10

    for _, tab in ipairs(self.Tabs) do
        local cy = contentStartY
        for _, ctrl in ipairs(tab.Controls) do
            local handler = Controls[ctrl.Type]
            if handler and handler.Layout then
                handler.Layout(self, ctrl, contentStartX, cy)
            end
            cy += Advance(ctrl)
        end
    end

    self.FooterText.Position = Vector2.new(x + math.floor(w * 0.5), y + h - self.FooterHeight)
end

function UI:Refresh()
    if not self.Ready then return end
    local vis = self.Visible

    Put(self, "ob", self.OuterBorder, "Visible", vis)
    Put(self, "bg", self.Background, "Visible", vis)
    Put(self, "ti", self.TitleText, "Visible", vis)
    Put(self, "oo", self.OwnerOuter, "Visible", vis)
    Put(self, "oi", self.OwnerInner, "Visible", vis)
    Put(self, "ol", self.OwnerLabelDraw, "Visible", vis)
    Put(self, "ck", self.ClockDraw, "Visible", vis)
    Put(self, "hl", self.HeaderLine, "Visible", vis)
    Put(self, "tl", self.TabLine, "Visible", vis)
    Put(self, "ft", self.FooterText, "Visible", vis)

    self.LastClock = nil -- force a fresh clock text when the menu is (re)shown
    self:TickClock()

    -- Tab strip (only the visible window is drawn)
    for i, btn in ipairs(self.TabButtons) do
        local show = vis and self:IsTabInWindow(i)
        local active = (i == self.ActiveTab)
        Put(btn, "bg", btn.Bg, "Visible", show)
        Put(btn, "bc", btn.Bg, "Color", active and Color3.fromRGB(35, 35, 35) or Color3.fromRGB(20, 20, 20))
        Put(btn, "in", btn.Indicator, "Visible", show and active)
        Put(btn, "lb", btn.Label, "Visible", show)
    end

    local canLeft = self.TabOffset > 0
    local canRight = self.TabOffset < (#self.Tabs - self.TabWindow)
    Put(self, "alb", self.ArrowLBg, "Visible", vis)
    Put(self, "arb", self.ArrowRBg, "Visible", vis)
    Put(self, "alt", self.ArrowLText, "Visible", vis)
    Put(self, "art", self.ArrowRText, "Visible", vis)
    Put(self, "alc", self.ArrowLText, "Color", canLeft and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(80, 80, 80))
    Put(self, "arc", self.ArrowRText, "Color", canRight and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(80, 80, 80))

    for tabIdx, tab in ipairs(self.Tabs) do
        local tabVisible = vis and (tabIdx == self.ActiveTab)
        -- a tab that is already hidden is not touched again (only the active tab does real work)
        if tabVisible or tab.Shown ~= false then
            for _, ctrl in ipairs(tab.Controls) do
                local handler = Controls[ctrl.Type]
                if handler and handler.Refresh then
                    handler.Refresh(self, ctrl, tabVisible)
                end
            end
            tab.Shown = tabVisible
        end
    end
end

-- Cheap refresh of the controls of the active tab only (used for live data such as the Security tab)
function UI:RefreshActiveTab()
    if not self.Ready or not self.Visible then return end
    local tab = self.Tabs[self.ActiveTab]
    if not tab then return end
    for _, ctrl in ipairs(tab.Controls) do
        local handler = Controls[ctrl.Type]
        if handler and handler.Refresh then
            handler.Refresh(self, ctrl, true)
        end
    end
end

-- Refreshes one control (e.g. the status line) without touching anything else
function UI:RefreshControl(ctrl)
    if not self.Ready then return end
    local handler = Controls[ctrl.Type]
    if handler and handler.Refresh then
        handler.Refresh(self, ctrl, self.Visible and self.ActiveTab == ctrl.TabIdx)
    end
end

function UI:SetVisible(state)
    local target = state == true
    if self.Visible == target then return end
    self.Visible = target
    if not target then
        self.Dragging = false
        self.ActiveSlider = nil
    end
    self:Refresh()
end

-- Moves the tab strip window by `delta` tabs. Returns true when it actually moved.
function UI:ScrollTabs(delta)
    if not self.Ready then return false end
    local maxOffset = math.max(0, #self.Tabs - self.TabWindow)
    local newOffset = math.clamp(self.TabOffset + delta, 0, maxOffset)
    if newOffset == self.TabOffset then return false end
    self.TabOffset = newOffset
    self.ActiveSlider = nil
    self:LayoutTabStrip()
    self:Refresh()
    return true
end

-- Scrolls the strip (if needed) so tab i is inside the visible window. Returns true if it scrolled.
function UI:EnsureTabVisible(i)
    local maxOffset = math.max(0, #self.Tabs - self.TabWindow)
    local newOffset = self.TabOffset
    if i < self.TabOffset + 1 then
        newOffset = i - 1
    elseif i > self.TabOffset + self.TabWindow then
        newOffset = i - self.TabWindow
    end
    newOffset = math.clamp(newOffset, 0, maxOffset)
    if newOffset ~= self.TabOffset then
        self.TabOffset = newOffset
        self:LayoutTabStrip()
        return true
    end
    return false
end

function UI:SetActiveTab(i)
    local tab = self.Tabs[i]
    if not tab or self.ActiveTab == i then return end
    self.ActiveTab = i
    self.ActiveSlider = nil
    self:EnsureTabVisible(i)
    if tab.OnShow then
        local ok, err = pcall(tab.OnShow)
        if not ok then
            warn("[0x] Tab OnShow error: " .. tostring(err))
        end
    end
    self:Refresh()
end

-- Mouse wheel anywhere over the menu scrolls the tab strip (debounced for touchpads)
function UI:OnWheel(mousePos, direction)
    if not self.Visible or not self.Ready then return end
    if not PointInRect(mousePos.X, mousePos.Y, self.Position.X, self.Position.Y, self.Width, self.Height) then
        return
    end
    local now = os.clock()
    if (now - self.LastWheel) < 0.07 then return end
    self.LastWheel = now
    self:ScrollTabs(direction > 0 and -1 or 1)
end

function UI:OnMouseDown(mousePos)
    if not self.Visible or not self.Ready then return end
    local mx, my = mousePos.X, mousePos.Y
    local x, y = self.Position.X, self.Position.Y

    -- Header Drag Check
    if PointInRect(mx, my, x, y, self.Width, self.HeaderHeight) then
        self.Dragging = true
        self.DragOffset = mousePos - self.Position
        return
    end

    -- Tab strip arrows
    local al, ar = self.ArrowL, self.ArrowR
    if al and PointInRect(mx, my, al.X, al.Y, al.W, al.H) then
        self:ScrollTabs(-1)
        return
    end
    if ar and PointInRect(mx, my, ar.X, ar.Y, ar.W, ar.H) then
        self:ScrollTabs(1)
        return
    end

    -- Tab Switch Check (visible window only)
    for i, btn in ipairs(self.TabButtons) do
        if self:IsTabInWindow(i) and PointInRect(mx, my, btn.X, btn.Y, btn.W, btn.H) then
            self:SetActiveTab(i)
            return
        end
    end

    -- Active Tab Controls Check
    local activeTab = self.Tabs[self.ActiveTab]
    if activeTab then
        for _, ctrl in ipairs(activeTab.Controls) do
            local handler = Controls[ctrl.Type]
            if handler and handler.MouseDown then
                if handler.MouseDown(self, ctrl, mx, my) then
                    return
                end
            end
        end
    end
end

function UI:OnMouseMove(mousePos)
    if not self.Visible or not self.Ready then return end

    if self.Dragging then
        local viewport = WorldState.ViewportSize
        local rawPos = mousePos - self.DragOffset
        local clampedX = math.clamp(math.floor(rawPos.X + 0.5), 0, math.max(0, viewport.X - self.Width))
        local clampedY = math.clamp(math.floor(rawPos.Y + 0.5), 0, math.max(0, viewport.Y - self.Height))
        local newPos = Vector2.new(clampedX, clampedY)
        if newPos ~= self.Position then
            self.Position = newPos
            self:UpdateLayout()
        end
        return
    end

    if self.ActiveSlider then
        Controls.slider.UpdateFromMouse(self, self.ActiveSlider, mousePos.X)
    end
end

function UI:OnMouseUp()
    local interacting = self.Dragging or self.ActiveSlider ~= nil
    self.Dragging = false
    self.ActiveSlider = nil
    if interacting then
        self:Changed()
    end
end

function UI:Destroy()
    self.Ready = false
    self.Dragging = false
    self.ActiveSlider = nil
    for _, obj in ipairs(self.Objects) do
        SafeRemoveDrawing(obj)
    end
    table.clear(self.Objects)
end

--==============================================================================
-- 15. Smart Staggered Dual-Point Wall-Check Raycast
--==============================================================================
local function CastPenetratingRay(data, cameraPos, targetPos, character, baseFilterCount)
    local filter = data.FilterTable
    -- Trim any extra penetrated parts from previous casts
    for i = #filter, baseFilterCount + 1, -1 do
        filter[i] = nil
    end

    local params = data.RayParams
    params.FilterDescendantsInstances = filter

    local currentOrigin = cameraPos

    for _ = 1, Config.WallPenetrationLimit do
        local direction = targetPos - currentOrigin
        local dist = direction.Magnitude
        if dist < 0.5 then
            return true
        end

        Metrics.RayCount += 1
        Metrics.FrameRays += 1
        local result = Workspace:Raycast(currentOrigin, direction, params)

        if not result then
            return true
        end

        local hitPart = result.Instance
        if not hitPart or hitPart:IsDescendantOf(character) then
            return true
        end

        if hitPart.Transparency > 0.5 or not hitPart.CanCollide then
            filter[#filter + 1] = hitPart
            params.FilterDescendantsInstances = filter
            currentOrigin = result.Position + (direction / dist) * 0.05
        else
            return false
        end
    end

    return false
end

local function CheckCharacterVisibility(data, now, cameraPos, headPos, rootPos, character, localChar)
    if (now - data.LastRayTime) < Config.RaycastInterval then
        return data.CachedVisible
    end
    -- Global per-frame budget: defer (without consuming the interval) when exhausted
    if Metrics.FrameRays >= Config.MaxRaysPerFrame then
        return data.CachedVisible
    end
    data.LastRayTime = now

    local filter = data.FilterTable
    filter[1] = WorldState.Camera
    local baseCount = 1
    if localChar then
        filter[2] = localChar
        baseCount = 2
    end

    -- Primary ray: Camera -> Head
    local visible = CastPenetratingRay(data, cameraPos, headPos, character, baseCount)

    -- Fallback ray: Camera -> Torso/RootPart (if head is behind a thin beam/sign)
    if not visible and rootPos then
        visible = CastPenetratingRay(data, cameraPos, rootPos, character, baseCount)
    end

    data.CachedVisible = visible
    return visible
end

local function HideSkeletonLines(data, fromIdx)
    local skelLines = data.SkeletonLines
    local lineShown = data.LineShown
    for i = fromIdx, 14 do
        if lineShown[i] then
            skelLines[i].Visible = false
            lineShown[i] = false
        end
    end
    if fromIdx == 1 then
        data.SkelShown = false
    end
end

local function HideCornerLines(data)
    if not data.CornerShown then return end
    data.CornerShown = false
    local cLines = data.CornerLines
    for i = 1, 8 do
        cLines[i].Visible = false
    end
end

local function HideArrow(data)
    if not data.ArrowShown then return end
    data.ArrowShown = false
    data.ArrowLines[1].Visible = false
    data.ArrowLines[2].Visible = false
end

local function HideHealth(data)
    if not data.HealthShown then return end
    data.HealthShown = false
    data.HealthOutline.Visible = false
    data.HealthFill.Visible = false
end

-- Hides everything attached to the character body (gated by the Hidden flag).
-- Off-screen arrows are managed independently (they are shown precisely while the body is hidden).
local function HideESPBody(data)
    if not data or data.Hidden or data.Destroyed then return end
    data.Hidden = true

    if data.BoxShown then
        data.Box.Visible = false
        data.BoxShown = false
    end
    if data.CornerShown then
        HideCornerLines(data)
    end
    if data.HeadShown then
        data.Head.Visible = false
        data.HeadShown = false
    end
    if data.TextShown then
        data.Text.Visible = false
        data.TextShown = false
    end
    if data.HealthShown then
        HideHealth(data)
    end
    if data.SkelShown then
        HideSkeletonLines(data, 1)
    end
end

-- Hides everything (body + arrow)
local function HideESPData(data)
    if not data or data.Destroyed then return end
    if data.ArrowShown then
        HideArrow(data)
    end
    HideESPBody(data)
end

local function HideAllESP()
    for _, data in pairs(ESPCache) do
        HideESPData(data)
    end
    Metrics.ActiveESPCount = 0
end

-- Draws the 2-line chevron pointing toward an off-screen enemy
local function PlaceArrow(data, cx, cy, dx, dy, radius, size, color, alpha)
    local tipX = cx + dx * radius
    local tipY = cy + dy * radius
    local backX = tipX - dx * size
    local backY = tipY - dy * size
    local px = -dy * size * 0.6
    local py = dx * size * 0.6

    local a = data.ArrowLines
    a[1].From = Vector2.new(math.floor(tipX + 0.5), math.floor(tipY + 0.5))
    a[1].To = Vector2.new(math.floor(backX + px + 0.5), math.floor(backY + py + 0.5))
    a[2].From = a[1].From
    a[2].To = Vector2.new(math.floor(backX - px + 0.5), math.floor(backY - py + 0.5))

    if data.ArrowColor ~= color or data.ArrowAlpha ~= alpha then
        data.ArrowColor = color
        data.ArrowAlpha = alpha
        a[1].Color = color
        a[1].Transparency = alpha
        a[2].Color = color
        a[2].Transparency = alpha
    end

    if not data.ArrowShown then
        a[1].Visible = true
        a[2].Visible = true
        data.ArrowShown = true
    end
end

-- Re-applies thickness / size to every drawing of one player (on config change)
local function ApplyStyle(data)
    data.StyleVersion = StyleVersion
    data.Box.Thickness = Config.BoxThickness
    data.Head.Thickness = Config.HeadThickness
    data.Text.Size = Config.TextSize
    local skelLines = data.SkeletonLines
    for i = 1, 14 do
        skelLines[i].Thickness = Config.SkeletonThickness
    end
    local cornerLines = data.CornerLines
    for i = 1, 8 do
        cornerLines[i].Thickness = Config.BoxThickness
    end
    data.LastColor = nil -- force color/alpha re-push
    data.ArrowColor = nil
    data.LastHp = -1
    data.TX = nil -- text offset depends on TextSize
end

-- Reads the Humanoid once and re-syncs the cached health (called at most once per second per player)
local function ResyncHealth(data, now)
    data.HpSyncAt = now + 1 + data.RayPhaseOffset
    local hum = data.Humanoid
    if not hum then return end
    local hp = hum.Health
    data.HpMax = hum.MaxHealth
    data.HpRatio = ComputeHpRatio(hp, data.HpMax)
    data.HpKnown = true
    data.Alive = type(hp) == "number" and hp > 0
end

local function ApplyRigProfile(data, rigType)
    if data.RigType == rigType then return end
    data.RigType = rigType

    if rigType == "R15" then
        data.NameMap = R15_NAME_TO_INDEX
        data.Bones = R15_BONES
        data.BoneCount = 14
        data.PartCount = 15
        data.LowestIndices = R15_LOWEST_INDICES
    elseif rigType == "R6" then
        data.NameMap = R6_NAME_TO_INDEX
        data.Bones = R6_BONES
        data.BoneCount = 5
        data.PartCount = 6
        data.LowestIndices = R6_LOWEST_INDICES
        HideSkeletonLines(data, 6)
    else
        data.NameMap = nil
        data.Bones = nil
        data.BoneCount = 0
        data.PartCount = 0
        data.LowestIndices = nil
    end

    -- Populate indexed PartArray from currently present children
    table.clear(data.PartArray)
    local char = data.Character
    local nameMap = data.NameMap
    if char and nameMap then
        for _, child in ipairs(char:GetChildren()) do
            if child:IsA("BasePart") then
                local idx = nameMap[child.Name]
                if idx then
                    data.PartArray[idx] = child
                end
            end
        end
    end
end

local function CreateESPObject(player)
    if player == LocalPlayer or ESPCache[player] or IsUnloaded or player.Parent == nil then
        return
    end

    local allocated = {}
    local function TrackDrawing(className, props)
        local obj = Drawing.new(className)
        table.insert(allocated, obj)
        for k, v in pairs(props) do
            obj[k] = v
        end
        return obj
    end

    PlayerSeedCounter += 1
    -- Golden-ratio phase offset to distribute raycasts smoothly across frames
    local rayPhaseOffset = (PlayerSeedCounter * 0.61803398875) % 1
    local startAlpha = Config.ColorHiddenAlpha

    local ok, data = pcall(function()
        local box = TrackDrawing("Square", {
            Thickness = Config.BoxThickness,
            Filled = false,
            Color = Config.ColorHidden,
            Transparency = startAlpha,
            ZIndex = 0,
            Visible = false,
        })

        local head = TrackDrawing("Circle", {
            Thickness = Config.HeadThickness,
            NumSides = HEAD_NUM_SIDES,
            Filled = false,
            Color = Config.ColorHidden,
            Transparency = startAlpha,
            ZIndex = 0,
            Visible = false,
        })

        local text = TrackDrawing("Text", {
            Size = Config.TextSize,
            Font = GetFont(),
            Center = true,
            Outline = true,
            Color = Config.ColorHidden,
            Transparency = startAlpha,
            ZIndex = 0,
            Visible = false,
        })

        local skeletonLines = table.create(14)
        local lineShown = table.create(14, false)
        for i = 1, 14 do
            skeletonLines[i] = TrackDrawing("Line", {
                Thickness = Config.SkeletonThickness,
                Color = Config.ColorHidden,
                Transparency = startAlpha,
                ZIndex = 0,
                Visible = false,
            })
        end

        local cornerLines = table.create(8)
        for i = 1, 8 do
            cornerLines[i] = TrackDrawing("Line", {
                Thickness = Config.BoxThickness,
                Color = Config.ColorHidden,
                Transparency = startAlpha,
                ZIndex = 0,
                Visible = false,
            })
        end

        -- Combat: health bar (outline + fill) and off-screen arrow (2 lines)
        local healthOutline = TrackDrawing("Square", {
            Size = Vector2.new(4, 4),
            Color = Color3.fromRGB(0, 0, 0),
            Filled = true,
            Transparency = 1,
            ZIndex = 0,
            Visible = false,
        })
        local healthFill = TrackDrawing("Square", {
            Size = Vector2.new(2, 2),
            Color = Color3.fromRGB(0, 255, 30),
            Filled = true,
            Transparency = 1,
            ZIndex = 1,
            Visible = false,
        })
        local arrowLines = table.create(2)
        for i = 1, 2 do
            arrowLines[i] = TrackDrawing("Line", {
                Thickness = 2,
                Color = Config.ColorHidden,
                Transparency = startAlpha,
                ZIndex = 0,
                Visible = false,
            })
        end

        local rayParams = RaycastParams.new()
        rayParams.FilterType = Enum.RaycastFilterType.Exclude
        rayParams.IgnoreWater = true
        rayParams.RespectCanCollide = false

        return {
            Player = player,
            Box = box,
            Head = head,
            Text = text,
            SkeletonLines = skeletonLines,
            CornerLines = cornerLines,
            HealthOutline = healthOutline,
            HealthFill = healthFill,
            ArrowLines = arrowLines,

            -- Reactive Character State
            Character = nil,
            Humanoid = nil,
            RootPart = nil,
            HeadPart = nil,
            Alive = false,

            -- Integer-Indexed Rig State
            RigType = nil,
            NameMap = nil,
            Bones = nil,
            BoneCount = 0,
            PartCount = 0,
            LowestIndices = nil,
            PartArray = table.create(15),
            JointScreen = table.create(15),

            -- Raycast & Style State
            FilterTable = table.create(8),
            RayParams = rayParams,
            RayPhaseOffset = rayPhaseOffset,
            LastRayTime = os.clock() - (rayPhaseOffset * Config.RaycastInterval),
            CachedVisible = false,
            LastColor = nil,
            LastAlpha = nil,
            StyleVersion = StyleVersion,
            LastDistanceInt = -1,
            LastHp = -1,
            ArrowColor = nil,
            ArrowAlpha = nil,

            -- Health cache (filled by Humanoid events + a 1s resync, never read per frame)
            HpRatio = 1,       -- target fraction 0..1
            HpShown = 1,       -- animated fraction actually drawn
            HpMax = 100,
            HpKnown = false,
            HpClock = os.clock(),
            HpSyncAt = 0,
            -- (BX..BH box, CX..CH corners, TX/TY text, HbX..HbH bar outline, HfX..HfH bar fill are created lazily:
            --  the last value written to the Drawing, so unchanged geometry costs zero bridge writes)

            -- Shadowed Drawing Visibility Flags
            Hidden = true,
            BoxShown = false,
            CornerShown = false,
            HeadShown = false,
            TextShown = false,
            SkelShown = false,
            HealthShown = false,
            ArrowShown = false,
            LineShown = lineShown,

            Bound = false,
            Destroyed = false,
            CharConnections = {},
        }
    end)

    if not ok or not data then
        Health.AllocFailures += 1
        Health.LastAllocFailAt = os.clock()
        for _, obj in ipairs(allocated) do
            SafeRemoveDrawing(obj)
        end
        return
    end

    ESPCache[player] = data
    Metrics.TrackedPlayers += 1

    local function UnbindHumanoid()
        local conns = data.CharConnections
        SafeDisconnect(conns.Died)
        SafeDisconnect(conns.HealthChanged)
        SafeDisconnect(conns.MaxHealthChanged)
        conns.Died = nil
        conns.HealthChanged = nil
        conns.MaxHealthChanged = nil
    end

    local function ResetHealthCache()
        data.HpRatio = 1
        data.HpShown = 1
        data.HpMax = 100
        data.HpKnown = false
        data.HpSyncAt = 0
        data.LastHp = -1
    end

    local function ClearCharacterState()
        UnbindHumanoid()
        SafeDisconnect(data.CharConnections.ChildAdded)
        SafeDisconnect(data.CharConnections.ChildRemoved)
        data.CharConnections.ChildAdded = nil
        data.CharConnections.ChildRemoved = nil

        data.Character = nil
        data.Humanoid = nil
        data.RootPart = nil
        data.HeadPart = nil
        data.Alive = false
        data.RigType = nil
        data.NameMap = nil
        data.Bones = nil
        data.BoneCount = 0
        data.PartCount = 0
        data.LowestIndices = nil
        ResetHealthCache()

        table.clear(data.PartArray)
        table.clear(data.JointScreen)
        table.clear(data.FilterTable) -- drops references to the previous character's parts
        data.CachedVisible = false
        HideESPData(data)
    end

    -- Single entry point for Humanoid health events
    local function OnHealth(hp)
        data.HpRatio = ComputeHpRatio(hp, data.HpMax)
        data.HpKnown = true
        local aliveNow = type(hp) == "number" and hp > 0
        data.Alive = aliveNow
        if not aliveNow then
            HideESPData(data)
        end
    end

    local function BindHumanoid(hum)
        if data.Humanoid == hum then return end
        UnbindHumanoid()

        data.Humanoid = hum
        data.HpMax = hum.MaxHealth
        data.HpShown = -1 -- forces a snap on the first drawn frame
        OnHealth(hum.Health)

        if not data.RigType then
            if hum.RigType == Enum.HumanoidRigType.R15 then
                ApplyRigProfile(data, "R15")
            elseif hum.RigType == Enum.HumanoidRigType.R6 then
                ApplyRigProfile(data, "R6")
            end
        end

        data.CharConnections.Died = hum.Died:Connect(function()
            data.Alive = false
            HideESPData(data)
        end)

        data.CharConnections.HealthChanged = hum.HealthChanged:Connect(OnHealth)

        data.CharConnections.MaxHealthChanged = hum:GetPropertyChangedSignal("MaxHealth"):Connect(function()
            data.HpMax = hum.MaxHealth
            data.HpRatio = ComputeHpRatio(hum.Health, data.HpMax)
        end)
    end

    local function InspectChildAdded(child)
        if data.Destroyed then return end
        if child:IsA("BasePart") then
            local name = child.Name
            if name == "HumanoidRootPart" then
                data.RootPart = child
            elseif name == "Head" then
                data.HeadPart = child
            end

            if not data.RigType then
                if name == "UpperTorso" or name == "LowerTorso" then
                    ApplyRigProfile(data, "R15")
                elseif name == "Torso" then
                    ApplyRigProfile(data, "R6")
                end
            end

            local nameMap = data.NameMap
            if nameMap then
                local idx = nameMap[name]
                if idx then
                    data.PartArray[idx] = child
                end
            end
        elseif child:IsA("Humanoid") then
            BindHumanoid(child)
        end
    end

    local function InspectChildRemoved(child)
        if data.Destroyed then return end
        if child == data.RootPart then
            data.RootPart = nil
        elseif child == data.HeadPart then
            data.HeadPart = nil
        elseif child == data.Humanoid then
            UnbindHumanoid()
            data.Humanoid = nil
            data.Alive = false
            data.HpKnown = false
        end

        local nameMap = data.NameMap
        if nameMap and child:IsA("BasePart") then
            local idx = nameMap[child.Name]
            if idx and data.PartArray[idx] == child then
                data.PartArray[idx] = nil
            end
        end
    end

    local function OnCharacterAdded(character)
        if not character or IsUnloaded or data.Destroyed then return end
        ClearCharacterState()

        data.Character = character

        -- Immediate synchronous pass over already-present instances
        for _, child in ipairs(character:GetChildren()) do
            InspectChildAdded(child)
        end

        -- Reactive listeners handle late-streamed parts & StreamingEnabled seamlessly
        data.CharConnections.ChildAdded = character.ChildAdded:Connect(InspectChildAdded)
        data.CharConnections.ChildRemoved = character.ChildRemoved:Connect(InspectChildRemoved)
    end

    data.Bind = function()
        if data.Bound or data.Destroyed or IsUnloaded then return end
        data.Bound = true
        data.CharConnections.Added = player.CharacterAdded:Connect(OnCharacterAdded)
        data.CharConnections.Removing = player.CharacterRemoving:Connect(ClearCharacterState)
        if player.Character then
            OnCharacterAdded(player.Character)
        end
    end

    data.Bind()
end

local function RemoveESPObject(player)
    local data = ESPCache[player]
    if not data then return end

    data.Destroyed = true

    for _, conn in pairs(data.CharConnections) do
        SafeDisconnect(conn)
    end
    table.clear(data.CharConnections)

    SafeRemoveDrawing(data.Box)
    SafeRemoveDrawing(data.Head)
    SafeRemoveDrawing(data.Text)
    SafeRemoveDrawing(data.HealthOutline)
    SafeRemoveDrawing(data.HealthFill)

    for i = 1, 14 do
        SafeRemoveDrawing(data.SkeletonLines[i])
    end
    for i = 1, 8 do
        SafeRemoveDrawing(data.CornerLines[i])
    end
    for i = 1, 2 do
        SafeRemoveDrawing(data.ArrowLines[i])
    end

    table.clear(data.PartArray)
    table.clear(data.JointScreen)
    table.clear(data.FilterTable)
    data.Character, data.Humanoid, data.RootPart, data.HeadPart = nil, nil, nil, nil
    data.Alive = false
    data.Bind = nil

    ESPCache[player] = nil
    Metrics.TrackedPlayers = math.max(0, Metrics.TrackedPlayers - 1)
end

-- Self-heal pass: creates objects for players that were missed (PlayerAdded error / alloc failure) and removes
-- objects of players that are gone (PlayerRemoving error). Returns the number of repairs.
local function ReconcilePlayers()
    if IsUnloaded then return 0 end
    local okList, list = pcall(Players.GetPlayers, Players)
    if not okList or type(list) ~= "table" then return 0 end

    local fixed = 0
    local present = {}
    for _, p in ipairs(list) do
        present[p] = true
        if p ~= LocalPlayer and not ESPCache[p] then
            CreateESPObject(p)
            if ESPCache[p] then fixed += 1 end
        end
    end

    local stale = nil
    for player, data in pairs(ESPCache) do
        if not present[player] or data.Destroyed then
            stale = stale or {}
            stale[#stale + 1] = player
        end
    end
    if stale then
        for _, player in ipairs(stale) do
            RemoveESPObject(player)
            fixed += 1
        end
    end

    -- counter drift can never survive a pass
    local n = 0
    for _ in pairs(ESPCache) do n += 1 end
    Metrics.TrackedPlayers = n

    if fixed > 0 then
        Health.Reconciled += fixed
        Health.LastReconcileAt = os.clock()
    end
    return fixed
end
Health.Reconcile = ReconcilePlayers

--==============================================================================
-- 17. Main RenderStepped Loop (Zero-Closure, State-Shadowed, Native-Optimized)
--==============================================================================
local function UpdateESP()
    if IsUnloaded then return end
    local espStartClock = os.clock()
    Metrics.FrameRays = 0

    local cam = Workspace.CurrentCamera
    if cam ~= WorldState.Camera then
        BindCamera(cam)
    end
    if not cam then
        HideAllESP()
        if Combat then
            pcall(Combat.Update, Combat, espStartClock, 0, 0, nil, math.huge, false, 0, 0)
        end
        return
    end

    local viewportPoint = cam.WorldToViewportPoint
    local viewportSize = WorldState.ViewportSize
    local screenW, screenH = viewportSize.X, viewportSize.Y
    local cameraCFrame = cam.CFrame
    local cameraPos = cameraCFrame.Position
    local cameraUp = cameraCFrame.UpVector
    local cameraRight = cameraCFrame.RightVector

    local localChar = WorldState.LocalChar
    local localRoot = WorldState.LocalRoot
    local originPos = (localRoot and localRoot.Position) or cameraPos
    local localTeam = LocalPlayer.Team

    local maxDist = Config.MaxDistance
    local maxDistSq = maxDist * maxDist

    local showBox = Config.ShowBox
    local useCorner = showBox and Config.CornerBox
    local useFullBox = showBox and not Config.CornerBox
    local showHead = Config.ShowHead
    local showDist = Config.ShowDistance
    local showSkel = Config.ShowSkeleton
    local visCheck = Config.VisibilityCheck
    local teamCheck = Config.TeamCheck
    local showHealth = Config.HealthBar
    local showArrows = Config.OffscreenArrows

    local colorVisible, colorHidden = Config.ColorVisible, Config.ColorHidden
    local alphaVisible, alphaHidden = Config.ColorVisibleAlpha, Config.ColorHiddenAlpha
    local styleVer = StyleVersion

    -- Off-screen arrow geometry (kept inside the screen even on small viewports)
    local centerX, centerY = screenW * 0.5, screenH * 0.5
    local arrowRadius = math.min(Config.ArrowRadius, math.max(40, math.min(screenW, screenH) * 0.5 - 24))
    local arrowSize = Config.ArrowSize

    -- Threat telemetry (feeds the Combat module)
    local threatCount, visibleCount = 0, 0
    local nearestData, nearestDistSq, nearestOn, nearestX, nearestY = nil, math.huge, false, 0, 0

    local activeCount = 0

    for player, data in pairs(ESPCache) do
        if teamCheck and localTeam ~= nil and player.Team == localTeam then
            HideESPData(data)
            continue
        end

        -- 1s staggered health resync (self-heals a missed Humanoid event)
        if espStartClock >= data.HpSyncAt and data.Humanoid then
            ResyncHealth(data, espStartClock)
        end

        local character = data.Character
        if not character or not character.Parent or not data.Alive then
            HideESPData(data)
            continue
        end

        -- custom rigs without a HumanoidRootPart fall back to the torso
        local rootPart = data.RootPart or data.PartArray[2]
        local headPart = data.HeadPart
        if not rootPart or not headPart then
            HideESPData(data)
            continue
        end

        local rootPos = rootPart.Position
        local delta = rootPos - originPos
        local distSq = delta:Dot(delta)
        if distSq > maxDistSq then
            HideESPData(data)
            continue
        end

        local rootScreen, rootOnScreen = viewportPoint(cam, rootPos)

        -- Threat telemetry (every valid enemy in range counts, on-screen or not)
        threatCount += 1
        if distSq < nearestDistSq then
            nearestDistSq = distSq
            nearestData = data
            nearestOn = rootOnScreen
            nearestX, nearestY = rootScreen.X, rootScreen.Y
        end

        -- Off-screen arrow (passive pointer toward enemies outside the view)
        if showArrows and not rootOnScreen then
            local rel = cameraCFrame:PointToObjectSpace(rootPos)
            local ax, ay = rel.X, -rel.Y
            local mag = math.sqrt(ax * ax + ay * ay)
            if mag < 1e-3 then
                ax, ay, mag = 0, 1, 1
            end
            PlaceArrow(data, centerX, centerY, ax / mag, ay / mag, arrowRadius, arrowSize, colorHidden, alphaHidden)
        elseif data.ArrowShown then
            HideArrow(data)
        end

        if rootScreen.Z <= 0 then
            HideESPBody(data)
            continue
        end

        if not rootOnScreen then
            if rootScreen.X < -OFFSCREEN_MARGIN or rootScreen.X > (screenW + OFFSCREEN_MARGIN)
                or rootScreen.Y < -OFFSCREEN_MARGIN or rootScreen.Y > (screenH + OFFSCREEN_MARGIN) then
                HideESPBody(data)
                continue
            end
        end

        local headPos = headPart.Position

        -- 0. Live style propagation (thickness / text size)
        if data.StyleVersion ~= styleVer then
            ApplyStyle(data)
        end

        -- 1. Visibility & Styling (Only updates Drawing properties on state change)
        local isVisible = false
        if visCheck then
            isVisible = CheckCharacterVisibility(data, espStartClock, cameraPos, headPos, rootPos, character, localChar)
        else
            data.CachedVisible = false
        end
        if isVisible then
            visibleCount += 1
        end

        local activeColor = isVisible and colorVisible or colorHidden
        local activeAlpha = isVisible and alphaVisible or alphaHidden

        if data.LastColor ~= activeColor or data.LastAlpha ~= activeAlpha then
            data.LastColor = activeColor
            data.LastAlpha = activeAlpha

            data.Box.Color = activeColor
            data.Box.Transparency = activeAlpha
            data.Head.Color = activeColor
            data.Head.Transparency = activeAlpha
            data.Text.Color = activeColor
            data.Text.Transparency = activeAlpha

            local skelLines = data.SkeletonLines
            for i = 1, 14 do
                skelLines[i].Color = activeColor
                skelLines[i].Transparency = activeAlpha
            end

            local cornerLines = data.CornerLines
            for i = 1, 8 do
                cornerLines[i].Color = activeColor
                cornerLines[i].Transparency = activeAlpha
            end
        end

        -- 2. Perspective-Accurate Bounding Box Calculation
        local needBounds = showBox or showDist or showHealth
        local boxX, boxY, boxW, boxH = 0, 0, 0, 0
        local validBounds = false

        if needBounds then
            local topWorld = headPos + Vector3.new(0, headPart.Size.Y * 0.5 + 0.35, 0)

            local lowestY = rootPos.Y - 2.75
            local lowestIndices = data.LowestIndices
            local partArray = data.PartArray
            if lowestIndices then
                local foundFoot = false
                local minFootY = math.huge
                for i = 1, #lowestIndices do
                    local p = partArray[lowestIndices[i]]
                    if p then
                        local bottomY = p.Position.Y - (p.Size.Y * 0.5)
                        if bottomY < minFootY then
                            minFootY = bottomY
                            foundFoot = true
                        end
                    end
                end
                if foundFoot then
                    lowestY = minFootY - 0.1
                end
            end

            local bottomWorld = Vector3.new(rootPos.X, lowestY, rootPos.Z)
            local topScreen = viewportPoint(cam, topWorld)
            local bottomScreen = viewportPoint(cam, bottomWorld)

            if topScreen.Z > 0 and bottomScreen.Z > 0 then
                local minY = math.min(topScreen.Y, bottomScreen.Y)
                local maxY = math.max(topScreen.Y, bottomScreen.Y)
                boxH = maxY - minY

                -- True perspective shoulder width so steep camera pitch never crushes box width
                local sideScreen = viewportPoint(cam, rootPos + cameraRight * 1.65)
                local perspW = math.abs(sideScreen.X - rootScreen.X) * 2

                boxH = math.max(boxH, perspW * 0.65)
                if boxH >= 4 then
                    boxW = math.max(boxH * 0.56, perspW)
                    boxX = math.floor(rootScreen.X - (boxW * 0.5))
                    boxY = math.floor(minY)
                    boxW = math.floor(boxW)
                    boxH = math.floor(boxH)
                    validBounds = true
                end
            end
        end

        -- 3. 2D Full Box vs Corner Box (State-Shadowed)
        if validBounds and useFullBox then
            if data.CornerShown then
                HideCornerLines(data)
            end
            if data.BW ~= boxW or data.BH ~= boxH then
                data.BW, data.BH = boxW, boxH
                data.Box.Size = Vector2.new(boxW, boxH)
            end
            if data.BX ~= boxX or data.BY ~= boxY then
                data.BX, data.BY = boxX, boxY
                data.Box.Position = Vector2.new(boxX, boxY)
            end
            if not data.BoxShown then
                data.Box.Visible = true
                data.BoxShown = true
            end
        elseif validBounds and useCorner then
            if data.BoxShown then
                data.Box.Visible = false
                data.BoxShown = false
            end

            local cLines = data.CornerLines
            if data.CX ~= boxX or data.CY ~= boxY or data.CW ~= boxW or data.CH ~= boxH then
                data.CX, data.CY, data.CW, data.CH = boxX, boxY, boxW, boxH

                local cw = math.max(3, math.floor(boxW * CORNER_FRACTION))
                local ch = math.max(3, math.floor(boxH * CORNER_FRACTION))
                local x1, y1 = boxX, boxY
                local x2, y2 = boxX + boxW, boxY + boxH

                cLines[1].From = Vector2.new(x1, y1)
                cLines[1].To   = Vector2.new(x1 + cw, y1)
                cLines[2].From = Vector2.new(x1, y1)
                cLines[2].To   = Vector2.new(x1, y1 + ch)

                cLines[3].From = Vector2.new(x2, y1)
                cLines[3].To   = Vector2.new(x2 - cw, y1)
                cLines[4].From = Vector2.new(x2, y1)
                cLines[4].To   = Vector2.new(x2, y1 + ch)

                cLines[5].From = Vector2.new(x1, y2)
                cLines[5].To   = Vector2.new(x1 + cw, y2)
                cLines[6].From = Vector2.new(x1, y2)
                cLines[6].To   = Vector2.new(x1, y2 - ch)

                cLines[7].From = Vector2.new(x2, y2)
                cLines[7].To   = Vector2.new(x2 - cw, y2)
                cLines[8].From = Vector2.new(x2, y2)
                cLines[8].To   = Vector2.new(x2, y2 - ch)
            end

            if not data.CornerShown then
                for i = 1, 8 do
                    cLines[i].Visible = true
                end
                data.CornerShown = true
            end
        else
            if data.BoxShown then
                data.Box.Visible = false
                data.BoxShown = false
            end
            if data.CornerShown then
                HideCornerLines(data)
            end
        end

        -- 3b. Health Bar (cached Humanoid health, animated fill, read-only)
        if validBounds and showHealth then
            local target = data.HpRatio
            local shown = data.HpShown
            local dt = espStartClock - data.HpClock
            data.HpClock = espStartClock
            if not data.HealthShown or dt > 0.25 or dt < 0 or shown < 0 then
                shown = target -- first frame / long gap: no animation
            else
                local diff = target - shown
                if diff > -0.004 and diff < 0.004 then
                    shown = target
                else
                    shown += diff * (1 - math.exp(-dt * 14))
                end
            end
            data.HpShown = shown

            local fillH = math.max(math.floor(boxH * shown + 0.5), 1)
            -- left of the box; flips to the right side when there is no room on the left edge of the screen
            local barX = boxX - 6
            if barX < 1 then
                barX = boxX + boxW + 3
            end
            local fillY = boxY + boxH - fillH

            if data.HbX ~= barX or data.HbY ~= boxY or data.HbH ~= boxH then
                data.HbX, data.HbY, data.HbH = barX, boxY, boxH
                data.HealthOutline.Size = Vector2.new(4, boxH + 2)
                data.HealthOutline.Position = Vector2.new(barX, boxY - 1)
            end
            if data.HfX ~= barX or data.HfY ~= fillY or data.HfH ~= fillH then
                data.HfX, data.HfY, data.HfH = barX, fillY, fillH
                data.HealthFill.Size = Vector2.new(2, fillH)
                data.HealthFill.Position = Vector2.new(barX + 1, fillY)
            end

            local hpInt = math.floor(shown * 100 + 0.5)
            if data.LastHp ~= hpInt then
                data.LastHp = hpInt
                data.HealthFill.Color = Color3.fromRGB(
                    math.floor(255 * math.min(1, 2 * (1 - shown))),
                    math.floor(255 * math.min(1, 2 * shown)),
                    30
                )
            end
            if not data.HealthShown then
                data.HealthOutline.Visible = true
                data.HealthFill.Visible = true
                data.HealthShown = true
            end
        elseif data.HealthShown then
            HideHealth(data)
        end

        -- 4. Distance Text (State-Shadowed)
        if validBounds and showDist then
            local distInt = math.floor(math.sqrt(distSq) + 0.5)
            if data.LastDistanceInt ~= distInt then
                data.LastDistanceInt = distInt
                data.Text.Text = string.format("%dm", distInt)
            end
            local textX, textY = boxX + math.floor(boxW * 0.5), boxY - (Config.TextSize + 4)
            if data.TX ~= textX or data.TY ~= textY then
                data.TX, data.TY = textX, textY
                data.Text.Position = Vector2.new(textX, textY)
            end
            if not data.TextShown then
                data.Text.Visible = true
                data.TextShown = true
            end
        elseif data.TextShown then
            data.Text.Visible = false
            data.TextShown = false
        end

        -- 5. Head ESP + Joint #1 Projection Cache
        local headVec2 = nil
        if showHead or (showSkel and data.BoneCount > 0) then
            local hScreen = viewportPoint(cam, headPos)
            if hScreen.Z > 0 then
                headVec2 = Vector2.new(hScreen.X, hScreen.Y)
            end
        end

        if showHead and headVec2 then
            local edgeWorld = headPos + (cameraUp * (headPart.Size.Y * 0.5 * Config.HeadPadding))
            local edgeScreen = viewportPoint(cam, edgeWorld)
            local dx = headVec2.X - edgeScreen.X
            local dy = headVec2.Y - edgeScreen.Y
            local radius = math.sqrt(dx * dx + dy * dy)

            data.Head.Position = headVec2
            data.Head.Radius = math.clamp(radius, 2, 200)
            if not data.HeadShown then
                data.Head.Visible = true
                data.HeadShown = true
            end
        elseif data.HeadShown then
            data.Head.Visible = false
            data.HeadShown = false
        end

        -- 6. Integer-Indexed Skeleton ESP (Deduplicated Joint Projection + State Shadowing)
        local boneCount = data.BoneCount
        if showSkel and boneCount > 0 then
            local partArray = data.PartArray
            local jointScreen = data.JointScreen
            local partCount = data.PartCount

            jointScreen[1] = headVec2
            for i = 2, partCount do
                local part = partArray[i]
                if part then
                    local pos = viewportPoint(cam, part.Position)
                    if pos.Z > 0 then
                        jointScreen[i] = Vector2.new(pos.X, pos.Y)
                    else
                        jointScreen[i] = nil
                    end
                else
                    jointScreen[i] = nil
                end
            end

            local bones = data.Bones
            local skelLines = data.SkeletonLines
            local lineShown = data.LineShown

            for i = 1, boneCount do
                local pair = bones[i]
                local pA = jointScreen[pair[1]]
                local pB = jointScreen[pair[2]]
                local line = skelLines[i]

                if pA and pB then
                    line.From = pA
                    line.To = pB
                    if not lineShown[i] then
                        line.Visible = true
                        lineShown[i] = true
                    end
                elseif lineShown[i] then
                    line.Visible = false
                    lineShown[i] = false
                end
            end

            data.SkelShown = true
        elseif data.SkelShown then
            HideSkeletonLines(data, 1)
        end

        data.Hidden = false
        activeCount += 1
    end

    -- 7. Combat module (isolated: a failure here never breaks the ESP frame)
    if Combat then
        pcall(Combat.Update, Combat, espStartClock, threatCount, visibleCount,
            nearestData, nearestDistSq, nearestOn, nearestX, nearestY)
    end

    -- 8. Performance Telemetry Sampling
    local now = os.clock()
    local espElapsedMs = (now - espStartClock) * 1000
    Metrics.FrameCount += 1
    Metrics.ESPSumMs += espElapsedMs
    if espElapsedMs > Metrics.ESPMaxSample then
        Metrics.ESPMaxSample = espElapsedMs
    end
    Metrics.ActiveESPCount = activeCount

    local sampleElapsed = now - Metrics.LastSampleTime
    if sampleElapsed >= 0.5 then
        local frames = math.max(1, Metrics.FrameCount)
        Metrics.FPS = math.floor((Metrics.FrameCount / sampleElapsed) + 0.5)
        Metrics.FrameTimeMs = (sampleElapsed / frames) * 1000
        Metrics.ESPAvgMs = Metrics.ESPSumMs / frames
        Metrics.ESPMaxMs = Metrics.ESPMaxSample
        Metrics.RaysPerSec = math.floor((Metrics.RayCount / sampleElapsed) + 0.5)

        Metrics.FrameCount = 0
        Metrics.RayCount = 0
        Metrics.ESPSumMs = 0
        Metrics.ESPMaxSample = 0
        Metrics.LastSampleTime = now

        if Stats and Stats.Visible then
            Stats:UpdateText()
        end
    end
end

-- Fault-tolerant frame wrapper: one bad frame (destroyed instance, executor hiccup, ...)
-- can never kill the loop. After repeated failures the ESP pauses for 2s, then self-heals.
-- The menu clock / Security tab keep running even while the ESP is paused.
local LoopErrors = 0
local NextMaintenance = 0
local function SafeFrame()
    if IsUnloaded then return end

    local now = os.clock()
    if now >= Health.PausedUntil then
        local ok, err = pcall(UpdateESP)
        if ok then
            LoopErrors = 0
        else
            LoopErrors += 1
            Health.LoopErrorsTotal += 1
            Health.LastLoopErrorAt = now
            if LoopErrors == 1 or LoopErrors % 60 == 0 then
                warn("[0x] Render loop error: " .. tostring(err))
            end
            if LoopErrors >= 30 then
                pcall(HideAllESP)
                Health.PausedUntil = now + 2
                Health.LoopPauses += 1
                LoopErrors = 0
            end
        end
    end

    -- Self-heal pass for the player table (every 3s) + frame-driven debounced auto-save
    if now >= NextMaintenance then
        NextMaintenance = now + 3
        pcall(ReconcilePlayers)
    end
    pcall(ConfigIO.Tick, now)

    -- Header clock (cheap: string work only once per second, and only while the menu is open)
    pcall(UI.TickClock, UI)
    -- Security tab live data (re-scan once per second, only while that tab is on screen)
    pcall(Security.Tick, Security, now)
end

--==============================================================================
-- 18. Unload & Complete Resource Teardown
--==============================================================================
local function Unload()
    if IsUnloaded then return end

    if Config.AutoSave then
        pcall(ConfigIO.Save) -- before teardown so the menu position is still valid
    end
    IsUnloaded = true
    ConfigIO.Dirty = false
    Health.Reconcile = nil

    -- Every stage is isolated: one failing step must never leave the rest leaking
    pcall(SafeDisconnect, WorldState.CameraConn)
    WorldState.CameraConn = nil
    for _, conn in pairs(WorldState.LocalCharConns) do
        pcall(SafeDisconnect, conn)
    end
    table.clear(WorldState.LocalCharConns)

    for _, conn in pairs(GlobalConnections) do
        pcall(SafeDisconnect, conn)
    end
    table.clear(GlobalConnections)

    for player in pairs(ESPCache) do
        pcall(RemoveESPObject, player)
    end
    table.clear(ESPCache)

    if Crosshair then pcall(Crosshair.Destroy, Crosshair) end
    if Combat then pcall(Combat.Destroy, Combat) end
    if UI then pcall(UI.Destroy, UI) end
    if Stats then pcall(Stats.Destroy, Stats) end

    table.clear(GuardWarnAt)

    if GlobalEnv.__0x_Unload == Unload then
        GlobalEnv.__0x_Unload = nil
    end
    if GlobalEnv.__EnemyESP_Unload == Unload then
        GlobalEnv.__EnemyESP_Unload = nil
    end
    if GlobalEnv.__0x_Token == Health.Token then
        GlobalEnv.__0x_Token = nil
    end

    local displayName = LocalPlayer.DisplayName or LocalPlayer.Name
    local goodbyeText = string.format("0x  |  GOODBYE, %s", string.upper(displayName))
    print(string.format("[0x] v%s (%s) Unloaded cleanly. Goodbye, %s!", VERSION, BUILD, displayName))
    ShowBanner(goodbyeText, Color3.fromRGB(255, 80, 80), 2.5)
end

Health.Unload = Unload
GlobalEnv.__0x_Unload = Unload
GlobalEnv.__EnemyESP_Unload = Unload
GlobalEnv.__0x_Token = Health.Token

local function ToggleKey(configKey, after)
    return function(state)
        Config[configKey] = Resolve(state, Config[configKey])
        if after then after() end
        UI:Refresh()
        ConfigIO.NotifyChanged()
        return Config[configKey]
    end
end

-- Read-only view of the live config. Writes are routed through the validated setter,
-- so external scripts can never push an out-of-range / wrong-type value into the engine.
local ConfigProxy = setmetatable({}, {
    __index = Config,
    __newindex = function(_, key, value)
        local ok, msg = ConfigIO.Set(key, value)
        if not ok then
            warn(string.format("[0x] Rejected Config.%s: %s", tostring(key), tostring(msg)))
        end
    end,
    __iter = function()
        return next, Config
    end,
    __metatable = "locked",
})
Health.ConfigProxy = ConfigProxy
